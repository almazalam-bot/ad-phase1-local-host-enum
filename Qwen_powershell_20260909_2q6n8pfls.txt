# UTILITY FUNCTIONS

function Write-TestHeader {
    param([string]$TestNumber, [string]$TestName)
    Write-Host "`n============================================================" -ForegroundColor Cyan
    Write-Host "TEST #$TestNumber : $TestName" -ForegroundColor Cyan
    Write-Host "============================================================" -ForegroundColor Cyan
    Write-Host "Timestamp: $(Get-Date -Format 'yyyy-MM-dd HH:mm:ss')" -ForegroundColor Yellow
}

function Write-TestResult {
    param([string]$Message, [string]$Color = "Green")
    Write-Host "[RESULT] $Message" -ForegroundColor $Color
}

function Wait-ForConfirmation {
    param([string]$TestName)
    Write-Host "`nPress ENTER to execute: $TestName" -ForegroundColor Yellow
    Write-Host "Press 'S' to SKIP this test" -ForegroundColor Yellow
    $key = $Host.UI.RawUI.ReadKey("NoEcho,IncludeKeyDown")
    if ($key.Character -eq 's' -or $key.Character -eq 'S') {
        Write-TestResult "SKIPPED" "Yellow"
        return $false
    }
    return $true
}

        Write-Host "  Task registration failed: $($_.Exception.Message)" -ForegroundColor DarkYellow

    Write-Host "  Creating task via schtasks.exe..." -ForegroundColor Gray
    schtasks /create /tn "QDB_Schtasks_Test" /tr "cmd.exe /c echo QDB-Schtasks-Test" /sc once /st (Get-Date).AddMinutes(2).ToString("HH:mm") /ru SYSTEM /f 2>&1 | Out-Null

    Write-TestResult "Scheduled task simulation completed. Check task creation/modification alerts."

    # Cleanup
